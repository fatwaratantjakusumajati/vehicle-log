"""
Sistem Deteksi Truk Cam1 - Versi Simplified (objektif-only)
==============================================================
cara akses venv = ./venv_311/Scripts/Activate.ps1

Perubahan dari versi "FIXED":
1. DIHAPUS: MIN_AREA_TRUCK & MIN_ASPECT_RATIO. Ini biang kerok kenapa
   pickup / mobil box sering tidak terdeteksi. Diganti satu filter noise
   opsional (MIN_AREA_NOISE) yang defaultnya sangat kecil / bisa 0.
2. DIHAPUS: TruckIdentityResolver versi lama yang mencocokkan ID pakai
   jarak+aspect+area (REMATCH_DIST_SHORT, X_CENTER_TOLERANCE,
   ASPECT_RATIO_TOLERANCE, AREA_TOLERANCE_RATIO, ENTRY_ZONE_BOTTOM_RATIO).
   Semua itu menduplikasi kerja yang SUDAH dilakukan oleh YOLO tracker
   sendiri (model.track(..., persist=True) sudah menghasilkan track_id
   yang stabil antar frame). Sekarang track_id YOLO dipakai langsung.
3. Fokus tetap: (a) deteksi truk/mobil masuk-keluar via crossing line,
   (b) status muatan saat masuk & keluar, (c) kirim ke n8n.
"""

import cv2
import numpy as np
import os
import time
import json
import base64
import urllib.request
from datetime import datetime
from ultralytics import YOLO

# --- KONFIGURASI ENVIRONMENT ---
os.environ["OPENCV_FFMPEG_CAPTURE_OPTIONS"] = (
    "rtsp_transport;tcp|reconnect;1|reconnect_streamed;1|reconnect_delay_max;5|stimeout;5000000"
)

# --- KONFIGURASI UTAMA ---
N8N_WEBHOOK = "http://localhost:5678/webhook/vehicle-log"
MODEL_PATH = "yolov8n_openvino_model"
# RTSP_URL = "rtsp://admin:Aristides@192.168.1.200:554/Streaming/Channels/102"
RTSP_URL = "Perekaman Layar 2026-07-20 140041.mp4"

# ROI CAM1 (Loading Kiri)
ROI_X_MIN, ROI_X_MAX = 0.183, 0.986
ROI_Y_MIN, ROI_Y_MAX = 0.194, 0.961

# Threshold Deteksi
CONFIDENCE_MIN = 0.15
IOU_THRESHOLD = 0.45
CLASSES_ALLOWED = [2, 7]  # car, truck. Kalau mobil box/bus sering keklasifikasi
                          # sebagai class 5 (bus), tambahkan 5 di sini setelah dicek COCO id.

# Filter noise MINIMAL saja (bukan filter "truk vs bukan truk").
# Set ke 0 kalau ingin benar-benar tanpa filter ukuran sama sekali.
MIN_AREA_NOISE = 0

# DEBUG
DEBUG_MODE = True

# Crossing Line
LINE_POSITION_RATIO = 0.55

# Track lifecycle
MAX_LOST_SECONDS = 3600   # 1 jam, sebelum track dianggap hilang & dibuang
EXIT_CLEANUP_SECONDS = 5  # setelah exit tercatat, tunggu sekian detik baru dibuang dari memory

# Inference
INFER_TARGET_WIDTH = 960  # harus kelipatan 32 (stride YOLO). Dipakai juga sebagai imgsz.

# Preprocessing
ENABLE_CLAHE = True
CLAHE_CLIP = 2.5
SAVE_JPEG_QUALITY = 50

# ---------------------------------------------------------------------
# FUNGSI PENDUKUNG UMUM
# ---------------------------------------------------------------------
def encode_image_base64(img, quality=SAVE_JPEG_QUALITY):
    ok, buf = cv2.imencode(".jpg", img, [cv2.IMWRITE_JPEG_QUALITY, quality])
    return base64.b64encode(buf).decode("utf-8") if ok else ""

def enhance_saved_crop(crop, max_dim=300):
    if crop is None or crop.size == 0:
        return crop
    h, w = crop.shape[:2]
    if max(h, w) < max_dim:
        scale = max_dim / max(h, w)
        crop = cv2.resize(crop, (max(1, int(w * scale)), max(1, int(h * scale))), interpolation=cv2.INTER_LINEAR)
    return crop

def prepare_roi_for_inference(roi_frame, target_width=INFER_TARGET_WIDTH):
    h, w = roi_frame.shape[:2]
    if w == 0 or h == 0:
        return roi_frame, 1.0
    scale = target_width / w
    if abs(scale - 1.0) < 0.02:
        return roi_frame, 1.0
    resized = cv2.resize(roi_frame, (target_width, int(round(h * scale))), interpolation=cv2.INTER_LINEAR)
    return resized, w / target_width

def apply_clahe(img_bgr):
    lab = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2LAB)
    l, a, b = cv2.split(lab)
    clahe = cv2.createCLAHE(clipLimit=CLAHE_CLIP, tileGridSize=(8, 8))
    l2 = clahe.apply(l)
    lab2 = cv2.merge((l2, a, b))
    return cv2.cvtColor(lab2, cv2.COLOR_LAB2BGR)

# ---------------------------------------------------------------------
# DETEKSI MUATAN (3 STATUS) - tidak diubah, ini memang core objektif
# ---------------------------------------------------------------------
CARGO_LOADED = "Bermuatan"
CARGO_EMPTY = "Kosong / bak tertutup"
CARGO_UNKNOWN = "Tidak dapat dipastikan"

def detect_cargo_status(truck_crop):
    if truck_crop is None or truck_crop.size < 5000:
        return CARGO_UNKNOWN, 0.0

    h, w = truck_crop.shape[:2]
    if h < 20 or w < 20:
        return CARGO_UNKNOWN, 0.0

    top_region = truck_crop[:int(h * 0.3), :]
    if top_region.size > 0:
        top_gray = cv2.cvtColor(top_region, cv2.COLOR_BGR2GRAY)
        top_edges = cv2.Canny(top_gray, 50, 150)
        edge_density = np.sum(top_edges > 0) / (top_edges.shape[0] * top_edges.shape[1])

        lines = cv2.HoughLinesP(top_edges, 1, np.pi / 180, threshold=50,
                                 minLineLength=int(w * 0.3), maxLineGap=10)

        if edge_density < 0.04 or (lines is not None and len(lines) > 2):
            return CARGO_EMPTY, 0.65

    bed_crop = truck_crop[int(h * 0.2):int(h * 0.8), int(w * 0.3):int(w * 0.95)]
    if bed_crop.size < 1000:
        return CARGO_UNKNOWN, 0.0

    gray = cv2.cvtColor(bed_crop, cv2.COLOR_BGR2GRAY)
    texture_var = cv2.Laplacian(gray, cv2.CV_64F).var()

    hsv = cv2.cvtColor(bed_crop, cv2.COLOR_BGR2HSV)
    hue_std = np.std(hsv[:, :, 0])
    value_std = np.std(hsv[:, :, 2])

    area_ratio = bed_crop.shape[0] * bed_crop.shape[1] / 20000.0
    threshold_texture = 300 * area_ratio
    threshold_color = 25 * area_ratio

    score = 0
    if texture_var > threshold_texture: score += 2
    if hue_std > threshold_color: score += 1.5
    if value_std > 25: score += 1

    if score >= 3:
        return CARGO_LOADED, 0.75
    elif score <= 0.5:
        return CARGO_EMPTY, 0.75
    else:
        return CARGO_UNKNOWN, 0.0

# ---------------------------------------------------------------------
# TRACK STATE - pakai track_id bawaan YOLO, tanpa re-matching manual
# ---------------------------------------------------------------------
class TruckTrackState:
    """
    Menyimpan status masuk/keluar per track_id yang sudah diberikan oleh
    YOLO tracker (persist=True). Tidak ada lagi pencocokan manual
    berdasarkan jarak/aspect-ratio/area -> itu tugas tracker, bukan tugas
    kode ini.
    """
    def __init__(self):
        self.tracks = {}  # track_id (str) -> data

    def get_or_create(self, track_id, timestamp):
        if track_id not in self.tracks:
            self.tracks[track_id] = {
                'entered': False,
                'entry_status': None,
                'exit_status': None,
                'exit_time': None,
                'last_seen': timestamp,
            }
        else:
            self.tracks[track_id]['last_seen'] = timestamp
        return self.tracks[track_id]

    def mark_entered(self, track_id, status):
        self.tracks[track_id]['entered'] = True
        self.tracks[track_id]['entry_status'] = status

    def finalize_exit(self, track_id, status, timestamp):
        data = self.tracks.get(track_id)
        if data:
            data['exit_status'] = status
            data['exit_time'] = timestamp
        return data

    def cleanup(self, timestamp):
        to_remove = []
        for tid, data in self.tracks.items():
            if data['exit_time'] and (timestamp - data['exit_time'] > EXIT_CLEANUP_SECONDS):
                to_remove.append(tid)
            elif not data['exit_time'] and (timestamp - data['last_seen'] > MAX_LOST_SECONDS):
                to_remove.append(tid)
        for tid in to_remove:
            self.tracks.pop(tid, None)

# ---------------------------------------------------------------------
# PENGIRIMAN KE N8N (FORMAT TETAP)
# ---------------------------------------------------------------------
def kirim_ke_n8n(payload):
    try:
        data = json.dumps(payload).encode()
        req = urllib.request.Request(N8N_WEBHOOK, data=data, headers={"Content-Type": "application/json"}, method="POST")
        urllib.request.urlopen(req, timeout=5)
        print(f"[N8N] Terkirim: {payload.get('jenis_kejadian')} - {payload.get('track_id')}")
    except Exception as e:
        print(f"[N8N] Gagal: {e}")

# ---------------------------------------------------------------------
# MAIN LOOP (SINGLE THREAD, SINGLE CAM)
# ---------------------------------------------------------------------
def main():
    cv2.setNumThreads(1)
    print("> Memuat YOLO Model (Cam1)...")
    model = YOLO(MODEL_PATH, task='detect')

    print("> Membuka RTSP Cam1...")
    # cap = cv2.VideoCapture(RTSP_URL, cv2.CAP_FFMPEG)
    cap = cv2.VideoCapture(RTSP_URL)
    cap.set(cv2.CAP_PROP_BUFFERSIZE, 1)

    tracks = TruckTrackState()
    frame_count = 0
    consecutive_fail = 0

    print("> Sistem Cam1 Mulai Berjalan (versi simplified)...")
    print(model.names)

    while True:
        ok = cap.grab()
        if not ok:
            consecutive_fail += 1
            if consecutive_fail > 10:
                print("[WARNING] Reconnecting RTSP...")
                cap.release()
                time.sleep(3)
                # cap = cv2.VideoCapture(RTSP_URL, cv2.CAP_FFMPEG)
                cap = cv2.VideoCapture(RTSP_URL)
                cap.set(cv2.CAP_PROP_BUFFERSIZE, 1)
                consecutive_fail = 0
            continue
        consecutive_fail = 0

        ret, frame = cap.retrieve()
        if not ret or frame is None:
            continue

        if frame_count == 0:
            print(f"[DEBUG] Frame pertama berhasil didapat! shape={frame.shape}")

        h_full, w_full = frame.shape[:2]
        rx1, ry1 = int(w_full * ROI_X_MIN), int(h_full * ROI_Y_MIN)
        rx2, ry2 = int(w_full * ROI_X_MAX), int(h_full * ROI_Y_MAX)
        roi_frame = frame[ry1:ry2, rx1:rx2].copy()
        if roi_frame.size == 0:
            continue
        roi_h, roi_w = roi_frame.shape[:2]
        if frame_count < 5:
            cv2.imwrite(f"debug_roi_{frame_count}.jpg", roi_frame)

        roi_resized, scale = prepare_roi_for_inference(roi_frame)
        if ENABLE_CLAHE:
            roi_resized = apply_clahe(roi_resized)

        results = model.track(
            roi_resized,
            imgsz=INFER_TARGET_WIDTH,
            conf=CONFIDENCE_MIN,
            iou=IOU_THRESHOLD,
            classes=CLASSES_ALLOWED,
            persist=True,
            verbose=False,
            tracker="bytetrack_custom.yaml",
        )
        result = results[0] if results else None

        now = time.time()
        frame_count += 1

        n_raw = len(result.boxes) if (result and result.boxes is not None) else 0
        has_id = bool(result and result.boxes is not None and result.boxes.id is not None)
        if DEBUG_MODE and frame_count % 15 == 0:
            if n_raw > 0:
                confs = [round(float(c), 3) for c in result.boxes.conf]
                print(f"[DEBUG] frame={frame_count} raw_detections={n_raw} has_track_id={has_id} confs={confs}")
            else:
                print(f"[DEBUG] frame={frame_count} raw_detections={n_raw} has_track_id={has_id}")

        if result and result.boxes is not None and result.boxes.id is not None:
            boxes = result.boxes
            ids = boxes.id.int().cpu().tolist()

            for i in range(len(boxes)):
                cls = int(boxes.cls[i])
                conf = float(boxes.conf[i])
                yolo_track_id = ids[i]
                x1r, y1r, x2r, y2r = boxes.xyxy[i].cpu().tolist()

                x1r, y1r, x2r, y2r = x1r * scale, y1r * scale, x2r * scale, y2r * scale
                width, height = x2r - x1r, y2r - y1r
                area = width * height

                if MIN_AREA_NOISE > 0 and area < MIN_AREA_NOISE:
                    if DEBUG_MODE:
                        print(f"[DEBUG] DIBUANG (noise) id={yolo_track_id} cls={cls} conf={conf:.2f} area={area:.0f}")
                    continue

                truck_id = f"TRK_{yolo_track_id:04d}"
                truck_data = tracks.get_or_create(truck_id, now)

                center_y = (y1r + y2r) / 2
                line_y = roi_h * LINE_POSITION_RATIO

                if DEBUG_MODE:
                    print(f"[DEBUG] {truck_id} cls={cls} conf={conf:.2f} area={area:.0f} "
                          f"center_y={center_y:.0f} line_y={line_y:.0f} entered={truck_data['entered']}")

                # --- CEK MASUK ---
                if not truck_data['entered']:
                    if center_y > line_y:
                        x1_c = int(max(0, x1r - 20)); y1_c = int(max(0, y1r - 20))
                        x2_c = int(min(roi_w, x2r + 20)); y2_c = int(min(roi_h, y2r + 20))
                        if x2_c > x1_c and y2_c > y1_c:
                            crop = roi_frame[y1_c:y2_c, x1_c:x2_c].copy()
                            crop_enhanced = enhance_saved_crop(crop)

                            status_muatan, _ = detect_cargo_status(crop_enhanced)
                            tracks.mark_entered(truck_id, status_muatan)

                            payload = {
                                "timestamp": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                                "kamera_id": "cam1",
                                "kamera_nama": "Loading Kiri",
                                "jenis_kendaraan": "Truk" if cls == 7 else "Mobil",
                                "warna": "Tidak Terdeteksi",
                                "confidence": round(float(conf), 4),
                                "bbox_x1": int(x1r), "bbox_y1": int(y1r),
                                "bbox_x2": int(x2r), "bbox_y2": int(y2r),
                                "rgb_r": 128, "rgb_g": 128, "rgb_b": 128,
                                "gambar_base64": encode_image_base64(crop_enhanced),
                                "status_muatan": status_muatan,
                                "jenis_kejadian": "MASUK",
                                "track_id": truck_id
                            }
                            kirim_ke_n8n(payload)

                # --- CEK KELUAR ---
                elif truck_data['entered'] and truck_data.get('exit_time') is None:
                    if center_y < line_y:
                        x1_c = int(max(0, x1r - 20)); y1_c = int(max(0, y1r - 20))
                        x2_c = int(min(roi_w, x2r + 20)); y2_c = int(min(roi_h, y2r + 20))
                        if x2_c > x1_c and y2_c > y1_c:
                            crop = roi_frame[y1_c:y2_c, x1_c:x2_c].copy()
                            crop_enhanced = enhance_saved_crop(crop)

                            status_muatan_keluar, _ = detect_cargo_status(crop_enhanced)
                            data_exit = tracks.finalize_exit(truck_id, status_muatan_keluar, now)

                            if data_exit:
                                payload = {
                                    "timestamp": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                                    "kamera_id": "cam1",
                                    "kamera_nama": "Loading Kiri",
                                    "jenis_kendaraan": "Truk" if cls == 7 else "Mobil",
                                    "warna": "Tidak Terdeteksi",
                                    "confidence": round(float(conf), 4),
                                    "bbox_x1": int(x1r), "bbox_y1": int(y1r),
                                    "bbox_x2": int(x2r), "bbox_y2": int(y2r),
                                    "rgb_r": 128, "rgb_g": 128, "rgb_b": 128,
                                    "gambar_base64": encode_image_base64(crop_enhanced),
                                    "status_muatan": status_muatan_keluar,
                                    "jenis_kejadian": "KELUAR",
                                    "track_id": truck_id
                                }
                                kirim_ke_n8n(payload)

        tracks.cleanup(now)

    cap.release()

if __name__ == "__main__":
    try:
        main()
    except KeyboardInterrupt:
        print("> Dihentikan oleh user.")
    except Exception as e:
        print(f"> FATAL ERROR: {e}")
